 360|      * config id) + {@code recharge_amount}. On a SUCCESS status the recharge amount
 361|      * is also credited to the customer wallet, mirroring the recharge-once series.</p>
 362|      *
 363|      * @param phone          10-digit customer mobile number
 364|      * @param configId       the selected AutoPay config id (the {@code autopay_id} of API-8)
 365|      * @param rechargeAmount the amount to be set up for AutoPay recharge
 366|      * @return step-by-step series result incl. the autopay_offers payload + wallet credit
 367|      */
 368|     public Map<String, Object> setupAutopay(String phone, String configId, String rechargeAmount) throws Exception {
 369|         return setupAutopay(phone, configId, rechargeAmount, null);
 370|     }

 372|     /**
 373|      * Overload that lets the caller force the AutoPay transaction status for QA/Demo
 374|      * (e.g. wish to observe a non-SUCCESS wallet behaviour without a real mandate).
 375|      *
 376|      * @param paymentStatus optional forced status (SUCCESS/PENDING/FAILED); null = poll real status.
 377|      */
 378|     public Map<String, Object> setupAutopay(String phone, String configId, String rechargeAmount, String paymentStatus) throws Exception {
 379|         if (phone == null || !phone.matches("\\\\d{10}")) {
 380|             throw new IllegalArgumentException("Enter a valid 10-digit mobile number");
 381|         }
 382|         if (configId == null || configId.trim().isEmpty()) {
 383|             throw new IllegalArgumentException("Select an AutoPay config");
 384|         }
 385|         String normalizedAmount = normalizeAmount(rechargeAmount);
 386|         String normalizedConfigId = configId.trim();
 387| 
 388|         String customerToken = resolveCustomerToken(phone);
 389|         int customerId = resolveCustomerId(phone);
 390| 
 391|         Map<String, Object> result = new HashMap<>();
 392|         result.put("phone", phone);
 393|         result.put("config_id", normalizedConfigId);
 394|         result.put("amount", normalizedAmount);
 395| 
 396|         // AutoPay saved-cards / recent payments run with payment_source "2" (the AutoPay
 397|         // source) \u2014 same value the recharge-once series uses. This also drives the
 398|         // cashback amount and the wallet balance before setup.
 399|         JsonNode savedCards = paymentApiClient.getSavedCardAndRecentPayments(customerToken, DEFAULT_PAYMENT_SOURCE);
 400|         result.put("saved_cards_response", savedCards);
 401|         String cashback = extractCashback(savedCards);
 402|         result.put("cashback", cashback);
 403| 
 404|         // API-8: fetch the smart-plan AutoPay offers for the selected config + amount.
 405|         JsonNode autopayOffers = paymentApiClient.getAutopayOffers(customerToken, normalizedConfigId, normalizedAmount);
 406|         result.put("autopay_offers", autopayOffers);
 407| 
 408|         String orderId = autopayOffers.path("data").path("order_id").asText(
 409|                 autopayOffers.path("data").path("merchant_transaction_id").asText(""));
 410|         result.put("order_id", orderId);
 411| 
 412|         Map<String, Object> status = new HashMap<>();
 413|         status.put("transaction_status", "PENDING_STARTED");
 414|         String txnStatusMsg = autopayOffers.path("data").path("status").asText("");
 415|         String finalStatus = "PENDING_STARTED";
 416|         boolean success = "SUCCESS".equalsIgnoreCase(txnStatusMsg)
 417|                 || "SUCCESSFUL".equalsIgnoreCase(txnStatusMsg)
 418|                 || "FINISHED".equalsIgnoreCase(txnStatusMsg)
 419|                 || "COMPLETED".equalsIgnoreCase(txnStatusMsg);
 420|         if (success) {
 421|             finalStatus = "SUCCESS";
 422|         }
 423| 
 424|         // Automatically credit the wallet on AutoPay setup success (mirror recharge-once).
 425|         Map<String, Object> walletCredit = null;
 426|         if (paymentStatus != null && !paymentStatus.trim().isEmpty()) {
 427|             String normalizedPaymentStatus = paymentStatus.trim().toUpperCase();
 428|             if (!("SUCCESS".equals(normalizedPaymentStatus) || "PENDING".equals(normalizedPaymentStatus)
 429|                     || "FAILED".equals(normalizedPaymentStatus))) {
 430|                 throw new IllegalArgumentException("paymentStatus must be SUCCESS, PENDING or FAILED");
 431|             }
 432|             success = "SUCCESS".equals(normalizedPaymentStatus);
 433|             finalStatus = normalizedPaymentStatus;
 434|             status.put("forced_status", true);
 435|         } else {
 436|             status.put("forced_status", false);
 437|         }
 438| 
 439|         if ("SUCCESS".equalsIgnoreCase(finalStatus)) {
 440|             walletCredit = dbUtil.applyWalletCredit(customerId, normalizedAmount, cashback,
 441|                     normalizedConfigId, DEFAULT_PAYMENT_METHOD, orderId);
 442|             result.put("wallet_balance_before", walletCredit.getOrDefault("old_wallet_balance", "0"));
 443|             result.put("updated_wallet_balance", walletCredit.get("new_wallet_balance"));
 444|         }
 445|         status.put("transaction_status", finalStatus);
 446|         result.put("forced_status", status.get("forced_status"));
 447|         result.put("wallet_credit", walletCredit);
 448|         return result;
 449|     }
